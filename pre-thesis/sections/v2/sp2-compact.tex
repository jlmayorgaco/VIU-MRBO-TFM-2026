\paragraph{E4 --- acoplamiento y transporte entre poses}
Con $\rho_i\in\Delta_3$ la mezcla avanzar/ceder/esperar de cada robot en un
conflicto de recurso $\ell$, $y_\ell(\rho)=\sum_{i,a}b_{ia\ell}\rho_{ia}$ y
$\beta_4,\alpha_4>0$,
\begin{equation}
 J_i(\rho_i,\rho_{-i})=\sum_ac_{ia}^{(4)}\rho_{ia}
 +\tfrac{\beta_4}{2}\sum_{\ell}y_\ell(\rho)^2+\tfrac{\alpha_4}{2}\|\rho_i\|_2^2.
 \label{eq:sp2-compact-e4-cost}
\end{equation}
Para la carga, con $e_k^L$ el error de pose y $W_k^d=-K_Pe_k^L-K_D\dot q_k^L$
la referencia de \emph{wrench},
\begin{equation}
 M_k\ddot q_k^L+D_k\dot q_k^L=G_{C_k}\boldsymbol\lambda_k^\star,\qquad
 \boldsymbol\lambda_k^\star\in\argmin_{0\leq\lambda\leq\bar\lambda}
 \|G_{C_k}\lambda-W_k^d\|_{Q_W}^2.
 \label{eq:sp2-compact-e4-pose}
\end{equation}

\begin{contribucion}{juego de vivacidad y servorregulación de pose}
\estadoaporte{demostrado por separado para la instantánea congelada y para contactos bilaterales fijos; demostraciones en el Anexo~\ref{app:sp4-game-proof} y el Anexo~\ref{app:sp4-pose-proof}.}
\begin{proposicion}[Potencial exacto y equilibrio variacional de E4]
\label{prop:sp2-compact-e4-potential}
Para costes y conflictos congelados, $\alpha_4>0$, simplex no vacíos, restricciones lineales $y_\ell\leq1$ y un punto interior factible (p. ej. la acción esperar satisface $y_\ell<1$ para todo $\ell$), los costes de~\eqref{eq:sp2-compact-e4-cost} definen un juego de potencial exacto cuyo minimizador coincide con los KKT del problema compartido y el equilibrio variacional de la relajación.
\end{proposicion}
\begin{teorema}[Disipación y estabilidad local de la pose]
\label{thm:sp2-compact-e4-pose}
Para la dinámica de la carga de la Ecuación~\eqref{eq:sp2-compact-e4-pose}, con $M_k,D_k,K_P,K_D\succ0$, objetivo y contactos fijos, y realización exacta del \emph{wrench} nominal ($\boldsymbol r_k^W=0$), la función $V=\tfrac12\dot q^{L\mathsf T}M_k\dot q^L+\tfrac12e_k^{L\mathsf T}K_Pe_k^L$ satisface $\dot V\leq0$; el principio de invariancia de LaSalle sobre el conjunto $\dot V=0$ da entonces estabilidad asintótica local del equilibrio de pose.
\end{teorema}
\end{contribucion}

\paragraph{E6 --- recuperación ante fallo}
Con $D_6(\boldsymbol x)$ el déficit residual y $K_6$ el coste de movilización,
\begin{equation}
 \Phi_6(\boldsymbol x)=-\lambda_6D_6(\boldsymbol x)-K_6(\boldsymbol x),\qquad
 U_i(x_i,\boldsymbol x_{-i})=\Phi_6(x_i,\boldsymbol x_{-i})-\Phi_6(0,\boldsymbol x_{-i}).
 \label{eq:sp2-compact-e6-potential}
\end{equation}

\begin{contribucion}{equilibrio de recuperación sujeto a certificado}
\estadoaporte{demostrado para el juego binario con déficit y costes congelados; demostraciones en el Anexo~\ref{app:sp6-proofs}.}
\begin{teorema}[Potencial exacto y terminación]
\label{thm:sp2-compact-e6-exact-potential}
El juego de~\eqref{eq:sp2-compact-e6-potential} es de potencial exacto; existe al menos un Nash puro y todo camino maximal de mejoras estrictas termina en, como máximo, $2^{n_R}-1$ cambios aceptados.
\end{teorema}
\begin{teorema}[Caracterización exacta de los Nash]
\label{thm:sp2-compact-e6-feasible-nash}
Con capacidades no negativas, costes positivos y $\lambda_6>\kappa_{\max}^6/\delta_{\min}$, los Nash puros coinciden exactamente con las coaliciones factibles mínimas por inclusión, de donde $\operatorname{PoS}_6=1$.
\end{teorema}
\begin{proposicion}[Ineficiencia no acotada del peor Nash]
\label{prop:sp2-compact-e6-unbounded-poa}
No existe cota uniforme, independiente de la instancia y de $\lambda_6$, para el cociente entre el peor Nash factible y el óptimo: $\operatorname{PoA}_6=+\infty$ incluso con un recurso y tres robots.
\end{proposicion}
\end{contribucion}

$\operatorname{PoS}_6=1$ garantiza un Nash óptimo cuando el tiempo no
restringe la reparación. La caída de éxito temporal bajo plazo estrecho es un
efecto empírico distinto, de distancia física entre robots y déficit, no una
instancia del contraejemplo de coste no acotado que prueba
$\operatorname{PoA}_6=+\infty$. El régimen de plazo estrecho, con sus cifras
e intervalos completos, se discute en el Anexo~\ref{app:sp2-full-detail}.

\paragraph{Cargo --- demostrador híbrido}
El demostrador ejecuta reclutamiento, acoplamiento, transporte y recuperación
sobre una misma planta planar con líder designado globalmente y reemplazo
voraz; sus reglas difieren de los mecanismos de E2, E3 y E6, y su evidencia
acredita compatibilidad funcional, no la composición formal de esas garantías.

\begin{algorithm}[!htbp]
\caption{Demostrador Cargo híbrido: selección, acoplamiento, transporte y sustitución}
\label{alg:cargo-hibrido-compact}
\begin{algorithmic}[1]
\Require identificadores activos, registro global, demanda agregada, guardia de seguridad
\Ensure entrega en pose destino, abstención, o fallo declarado con su causa
\State propagar identificadores (7 eventos, 10 si la red está degradada) y ordenar candidatos por $s_i=\bar\delta_i-0.08\bar c_i-0.025\bar f_i$
\While{no se satisface el criterio agregado $|C_k|\geq3$, $\sum c_i^{\mathrm{pay}}\geq m_k^{\mathrm{req}}$, $\sum f_i^{\max}\geq F_k^{\mathrm{req}}$ y quedan candidatos}
  \State añadir el siguiente candidato
\EndWhile
\If{no se satisface el criterio}
  \State \Return abstención
\EndIf
\State acoplar uniciclos; transportar con punto intermedio y filtro de seguridad
\If{falla el miembro de menor límite de fuerza}
  \State detener, retirar identificador, seleccionar y desplazar reemplazo
  \If{llega y el criterio vuelve a cumplirse}
    \State reanudar
  \Else
    \State \Return fallo
  \EndIf
\EndIf
\State \Return entrega en pose destino
\end{algorithmic}
\end{algorithm}

El demostrador alcanzó una tasa de misión $\CargoEtwoESuccess$ sin
intersección durante transporte, frente a $\CargoEtwoECentralSuccess$ de la
referencia central. En los tres regímenes de riesgo (obstáculo estático, red

este contraste), la guardia física cambió el éxito en
$\CargoEtwoEGuardEffect$; continuar con reemplazo tras un fallo lo elevó en
$\CargoEtwoERepairEffect$ frente a detenerse. El Anexo~\ref{app:sp2-full-detail}
detalla el protocolo, los cuatro regímenes y las ablaciones.
