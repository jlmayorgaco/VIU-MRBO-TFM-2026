% ---------------------------------------------------------------------
% Tipos de columna de los consolidados. En el original son L{}, C{} e Y,
% pero `C` e `Y` ya existen en viu-mrob-thesis.sty como columnas X sin
% argumento. Se renombran a J{}, K{} y Z, que están libres. Tienen que ser
% de una sola letra: `array` no maneja bien los nombres de varias y acaba
% rompiendo la `Y` de las tablas que ya existían.
% ---------------------------------------------------------------------
\newcolumntype{J}[1]{>{\RaggedRight\arraybackslash}p{#1}}
\newcolumntype{K}[1]{>{\centering\arraybackslash}p{#1}}
\newcolumntype{Z}{>{\RaggedRight\arraybackslash}X}

% ---------------------------------------------------------------------
% Puntos y leyenda del mapa metodológico académico, portados desde
% `thesis-reviews/academic_literature_review (1).tex` (líneas 60-68).
% #1,#2 = coordenadas; #3 = autor; #4 = año; #5 = inicial; #6 = número de
% referencia; #7 = color de familia.

\newcommand{\mappointL}[7]{%
  \filldraw[fill=#7,draw=white,line width=.64pt] (#1,#2) circle (2.38mm);
  \node[text=white,font=\sffamily\bfseries\fontsize{5.55}{5.75}\selectfont] at (#1,#2){#5};
  \node[anchor=west,align=left,text=#7!70!black,font=\sffamily\bfseries\fontsize{5.35}{5.60}\selectfont] at ($(#1,#2)+(0.036,0)$) {#3\,\textsuperscript{[#6]}\\[-.45mm]\fontsize{4.68}{4.95}\selectfont #4};}
\newcommand{\mappointR}[7]{%
  \filldraw[fill=#7,draw=white,line width=.64pt] (#1,#2) circle (2.38mm);
  \node[text=white,font=\sffamily\bfseries\fontsize{5.55}{5.75}\selectfont] at (#1,#2){#5};
  \node[anchor=east,align=right,text=#7!70!black,font=\sffamily\bfseries\fontsize{5.35}{5.60}\selectfont] at ($(#1,#2)+(-0.036,0)$) {#3\,\textsuperscript{[#6]}\\[-.45mm]\fontsize{4.68}{4.95}\selectfont #4};}

% ---------------------------------------------------------------------
% Celda de sistema de la matriz industrial. El original reserva un hueco
% de fotografía enlazado a la fuente oficial; aquí se conserva el enlace y
% se sustituye el hueco por el número de caso, que es lo que la memoria
% necesita para casar la tabla con el mapa.
% ---------------------------------------------------------------------
% El círculo se centra en (3.1,3.1), no en el origen: con la caja declarada en
% (0,0)-(6.2,6.2) y el trazo centrado en (0,0) el disco sobresalía 1 mm por la
% izquierda y la celda desbordaba por 2,85 pt en cada fila de la matriz.
\newcommand{\casenum}[1]{%
  \begin{tikzpicture}[x=1mm,y=1mm,baseline=-0.7mm]
    \path[use as bounding box] (0,0) rectangle (6.2,6.2);
    \fill[autumn] (3.1,3.1) circle (3.0);
    \node[font=\sffamily\bfseries\fontsize{5.4}{5.6}\selectfont,text=white] at (3.1,3.1) {#1};
  \end{tikzpicture}%
}
\newcommand{\matrixsystem}[5]{%
  \begin{minipage}[t]{\linewidth}\vspace{0pt}%
  \begin{minipage}[c]{6.4mm}\centering\casenum{#1}\end{minipage}%
  \hspace{0.4mm}%
  % \sloppy y las penalizaciones bajas dejan partir nombres como «Dual-AMR» o
  % «SL/HL-AGV», que si no desbordan la celda por unos pocos puntos.
  \begin{minipage}[c]{26.4mm}\raggedright\sloppy\hyphenpenalty=50\exhyphenpenalty=50\relax
    {\sffamily\bfseries\href{#5}{#2}}\par
    {\fontsize{5.35}{5.65}\selectfont\color{muted}#3 · #4}
  \end{minipage}%
  \end{minipage}%
}
